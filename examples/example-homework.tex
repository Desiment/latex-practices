\section*{Задачи}

\begin{exercise}
	Вычислите неопределённый интеграл $\int x \, dx$.
\end{exercise}

\begin{tasks}(2)
	\task $\int x^2 \, dx$
	\task $\int 1 \, dx$
\end{tasks}

\begin{solution}
	\begin{enumerate}
		\item $\int x^2 \, dx = \frac{x^3}{3} + C$
		\item $\int 1 \, dx = x + C$
	\end{enumerate}
\end{solution}
